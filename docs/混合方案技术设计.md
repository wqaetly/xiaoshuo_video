# 小说转视频 - 混合方案技术设计

> 版本: 1.0
> 方案: 本地LLM + 本地图像 + 本地TTS + 远端视频API
> 硬件: RTX 5080 (16GB VRAM)

---

## 1. 技术栈确定

### 1.1 最终选型

| 模块 | 技术选择 | 部署方式 | 成本 |
|------|----------|----------|------|
| **LLM分镜** | Ollama + Qwen2.5-14B | 本地 | 免费 |
| **图像生成** | ComfyUI + Z-Image-Turbo (6B) | 本地 | 免费 |
| **视频生成** | 即梦AI API / 可灵AI API | 远端API | 按量付费 |
| **TTS配音** | CosyVoice / Edge TTS | 本地 | 免费 |
| **视频合成** | FFmpeg + MoviePy | 本地 | 免费 |
| **字幕生成** | 直接生成 (基于TTS时间戳) | 本地 | 免费 |

> **Z-Image-Turbo**: 阿里通义实验室6B参数模型，仅需4步生成，中文原生支持

### 1.2 成本预估

```
固定成本: ¥0/月 (本地部署)
可变成本: 视频API调用

```

## 5. 核心模块设计

### 5.1 Pipeline控制器

```python
# src/pipeline/controller.py

from enum import Enum
from dataclasses import dataclass
from typing import Optional, Callable
import json
from pathlib import Path

class Phase(Enum):
    INIT = "init"
    ANALYZE = "analyze"
    CHARACTER_DESIGN = "character_design"
    GENERATE_IMAGES = "generate_images"
    GENERATE_AUDIO = "generate_audio"
    GENERATE_VIDEO = "generate_video"
    COMPOSE = "compose"
    DONE = "done"

@dataclass
class PipelineState:
    current_phase: Phase
    current_scene_index: int
    total_scenes: int
    errors: list

    def save(self, path: Path):
        with open(path, 'w', encoding='utf-8') as f:
            json.dump({
                'phase': self.current_phase.value,
                'scene_index': self.current_scene_index,
                'total_scenes': self.total_scenes,
                'errors': self.errors
            }, f)

    @classmethod
    def load(cls, path: Path) -> 'PipelineState':
        with open(path, 'r', encoding='utf-8') as f:
            data = json.load(f)
        return cls(
            current_phase=Phase(data['phase']),
            current_scene_index=data['scene_index'],
            total_scenes=data['total_scenes'],
            errors=data['errors']
        )

class PipelineController:
    """主流程控制器"""

    def __init__(self, project_path: Path):
        self.project_path = project_path
        self.state_file = project_path / "pipeline_state.json"
        self.state: Optional[PipelineState] = None

        # 加载各模块
        self.llm = None
        self.image_gen = None
        self.video_gen = None
        self.tts = None
        self.composer = None

        # 回调
        self.on_progress: Optional[Callable] = None

    def init_modules(self):
        """初始化各个模块"""
        from ..llm import OllamaClient, StoryboardGenerator
        from ..image import ComfyUIClient, SceneGenerator
        from ..video import VideoAPIClient
        from ..tts import CosyVoiceClient
        from ..compose import VideoComposer

        config = self._load_config()

        self.llm = OllamaClient(config['local']['ollama_url'])
        self.storyboard_gen = StoryboardGenerator(self.llm)
        self.image_gen = SceneGenerator(
            ComfyUIClient(config['local']['comfyui_url'])
        )
        self.video_gen = VideoAPIClient(
            provider=config['api']['video_provider'],
            api_key=config['api']['video_api_key']
        )
        self.tts = CosyVoiceClient(config['local']['cosyvoice_url'])
        self.composer = VideoComposer()

    def run(self, resume: bool = True):
        """执行完整流程"""
        # 检查是否可以断点续传
        if resume and self.state_file.exists():
            self.state = PipelineState.load(self.state_file)
            print(f"从 {self.state.current_phase.value} 阶段恢复")
        else:
            self.state = PipelineState(
                current_phase=Phase.INIT,
                current_scene_index=0,
                total_scenes=0,
                errors=[]
            )

        # 执行各阶段
        try:
            if self.state.current_phase == Phase.INIT:
                self._phase_init()

            if self.state.current_phase == Phase.ANALYZE:
                self._phase_analyze()

            if self.state.current_phase == Phase.CHARACTER_DESIGN:
                self._phase_character_design()

            if self.state.current_phase == Phase.GENERATE_IMAGES:
                self._phase_generate_images()

            if self.state.current_phase == Phase.GENERATE_AUDIO:
                self._phase_generate_audio()

            if self.state.current_phase == Phase.GENERATE_VIDEO:
                self._phase_generate_video()

            if self.state.current_phase == Phase.COMPOSE:
                self._phase_compose()

            self.state.current_phase = Phase.DONE
            self._save_state()
            print("✅ 视频生成完成!")

        except Exception as e:
            self.state.errors.append(str(e))
            self._save_state()
            raise

    def _phase_init(self):
        """初始化阶段"""
        print("📁 初始化项目...")
        self.init_modules()
        self._ensure_directories()
        self.state.current_phase = Phase.ANALYZE
        self._save_state()

    def _phase_analyze(self):
        """分析阶段 - 使用本地LLM"""
        print("🔍 分析小说内容...")

        novel_path = self.project_path / "input" / "novel.txt"
        novel_text = novel_path.read_text(encoding='utf-8')

        # 1. 提取角色
        characters = self.storyboard_gen.extract_characters(novel_text)
        self._save_json("characters.json", characters)

        # 2. 生成分镜
        storyboard = self.storyboard_gen.generate(novel_text, characters)
        self._save_json("storyboard.json", storyboard)

        self.state.total_scenes = len(storyboard['scenes'])
        self.state.current_phase = Phase.CHARACTER_DESIGN
        self._save_state()
        self._report_progress("分析完成", f"共{self.state.total_scenes}个场景")

    def _phase_character_design(self):
        """角色设计阶段 - 本地ComfyUI"""
        print("🎨 生成角色立绘...")

        characters = self._load_json("characters.json")

        for char in characters['characters']:
            if self._character_exists(char['id']):
                continue

            # 生成角色图像
            images = self.image_gen.generate_character(char)

            # 保存
            char_dir = self.project_path / "characters" / char['id']
            char_dir.mkdir(parents=True, exist_ok=True)
            for name, img in images.items():
                img.save(char_dir / f"{name}.png")

        self.state.current_phase = Phase.GENERATE_IMAGES
        self._save_state()

    def _phase_generate_images(self):
        """图像生成阶段 - 本地ComfyUI"""
        print("🖼️ 生成场景图像...")

        storyboard = self._load_json("storyboard.json")
        characters = self._load_json("characters.json")

        for i, scene in enumerate(storyboard['scenes']):
            if i < self.state.current_scene_index:
                continue

            if scene['generation_status']['image'] == 'done':
                continue

            # 生成图像
            image = self.image_gen.generate_scene(scene, characters)

            # 保存
            image_path = self.project_path / "images" / f"{scene['id']}.png"
            image.save(image_path)

            # 更新状态
            scene['generation_status']['image'] = 'done'
            self.state.current_scene_index = i + 1
            self._save_state()
            self._save_json("storyboard.json", storyboard)
            self._report_progress("图像生成", f"{i+1}/{self.state.total_scenes}")

        self.state.current_scene_index = 0
        self.state.current_phase = Phase.GENERATE_AUDIO
        self._save_state()

    def _phase_generate_audio(self):
        """音频生成阶段 - 本地CosyVoice"""
        print("🎙️ 生成配音...")

        storyboard = self._load_json("storyboard.json")
        characters = self._load_json("characters.json")

        for i, scene in enumerate(storyboard['scenes']):
            if scene['generation_status']['audio'] == 'done':
                continue

            # 生成配音
            audio_data = self.tts.generate_scene_audio(
                scene['audio'],
                characters
            )

            # 保存
            audio_path = self.project_path / "audio" / f"{scene['id']}.wav"
            audio_data.save(audio_path)

            # 更新状态
            scene['generation_status']['audio'] = 'done'
            scene['audio']['duration'] = audio_data.duration
            self._save_json("storyboard.json", storyboard)
            self._report_progress("音频生成", f"{i+1}/{self.state.total_scenes}")

        self.state.current_phase = Phase.GENERATE_VIDEO
        self._save_state()

    def _phase_generate_video(self):
        """视频生成阶段 - 远端API"""
        print("🎬 生成视频片段 (调用API)...")

        storyboard = self._load_json("storyboard.json")

        for i, scene in enumerate(storyboard['scenes']):
            if scene['generation_status']['video'] == 'done':
                continue

            image_path = self.project_path / "images" / f"{scene['id']}.png"

            # 调用远端API生成视频
            video_data = self.video_gen.generate(
                image_path=image_path,
                motion_prompt=self._build_motion_prompt(scene),
                duration=scene['duration']
            )

            # 保存
            video_path = self.project_path / "videos" / f"{scene['id']}.mp4"
            video_data.save(video_path)

            # 更新状态
            scene['generation_status']['video'] = 'done'
            self._save_json("storyboard.json", storyboard)
            self._report_progress("视频生成", f"{i+1}/{self.state.total_scenes}")

        self.state.current_phase = Phase.COMPOSE
        self._save_state()

    def _phase_compose(self):
        """合成阶段 - 本地FFmpeg"""
        print("🎞️ 合成最终视频...")

        storyboard = self._load_json("storyboard.json")

        # 收集所有片段
        clips = []
        for scene in storyboard['scenes']:
            clips.append({
                'video': self.project_path / "videos" / f"{scene['id']}.mp4",
                'audio': self.project_path / "audio" / f"{scene['id']}.wav",
                'subtitle': scene['subtitle']
            })

        # 合成
        output_path = self.project_path / "output" / "final_video.mp4"
        self.composer.compose(
            clips=clips,
            output_path=output_path,
            bgm_path=self._get_bgm_path()
        )

        self._report_progress("完成", str(output_path))

    # 辅助方法
    def _save_state(self):
        self.state.save(self.state_file)

    def _save_json(self, name: str, data: dict):
        path = self.project_path / name
        with open(path, 'w', encoding='utf-8') as f:
            json.dump(data, f, ensure_ascii=False, indent=2)

    def _load_json(self, name: str) -> dict:
        path = self.project_path / name
        with open(path, 'r', encoding='utf-8') as f:
            return json.load(f)

    def _load_config(self) -> dict:
        import yaml
        config_path = self.project_path / "project.yaml"
        with open(config_path, 'r', encoding='utf-8') as f:
            return yaml.safe_load(f)

    def _ensure_directories(self):
        for dir_name in ['characters', 'images', 'videos', 'audio', 'output']:
            (self.project_path / dir_name).mkdir(parents=True, exist_ok=True)

    def _character_exists(self, char_id: str) -> bool:
        char_dir = self.project_path / "characters" / char_id
        return char_dir.exists() and any(char_dir.iterdir())

    def _build_motion_prompt(self, scene: dict) -> str:
        camera = scene['visual']['camera']
        return f"{camera['type']}, from {camera['start_frame']} to {camera['end_frame']}"

    def _get_bgm_path(self):
        # 返回背景音乐路径
        return None

    def _report_progress(self, stage: str, detail: str):
        if self.on_progress:
            self.on_progress(stage, detail)
```

from typing import List, Dict
import json
import re

class StoryboardGenerator:
    """分镜脚本生成器"""

    def __init__(self, llm_client):
        self.llm = llm_client
        self.prompts = self._load_prompts()

    def extract_characters(self, novel_text: str) -> Dict:
        """提取角色信息"""
        prompt = self.prompts['extract_characters'].format(
            novel_text=novel_text[:10000]  # 限制长度
        )

        response = self.llm.chat(prompt)
        characters = self._parse_json_response(response)

        return {"characters": characters}

    def generate(self, novel_text: str, characters: Dict) -> Dict:
        """生成完整分镜脚本"""

        # 分章处理
        chapters = self._split_chapters(novel_text)
        all_scenes = []

        for chapter_idx, chapter_text in enumerate(chapters):
            scenes = self._generate_chapter_scenes(
                chapter_text,
                chapter_idx + 1,
                characters
            )
            all_scenes.extend(scenes)

        return {
            "novel_title": self._extract_title(novel_text),
            "total_scenes": len(all_scenes),
            "characters": characters['characters'],
            "scenes": all_scenes
        }

    def _generate_chapter_scenes(
        self,
        chapter_text: str,
        chapter_num: int,
        characters: Dict
    ) -> List[Dict]:
        """为单个章节生成分镜"""

        prompt = self.prompts['generate_storyboard'].format(
            chapter_text=chapter_text,
            chapter_num=chapter_num,
            characters_json=json.dumps(characters, ensure_ascii=False)
        )

        response = self.llm.chat(prompt)
        scenes = self._parse_json_response(response)

        # 添加序号和状态
        for i, scene in enumerate(scenes):
            scene['id'] = f"scene_{chapter_num:02d}_{i+1:03d}"
            scene['chapter'] = chapter_num
            scene['sequence'] = i + 1
            scene['generation_status'] = {
                'image': 'pending',
                'video': 'pending',
                'audio': 'pending'
            }

        return scenes

    def _split_chapters(self, text: str) -> List[str]:
        """分割章节"""
        # 匹配常见章节格式
        pattern = r'第[一二三四五六七八九十百千\d]+章'
        parts = re.split(pattern, text)
        return [p.strip() for p in parts if p.strip()]

    def _parse_json_response(self, response: str) -> any:
        """从LLM响应中提取JSON"""
        # 尝试找到JSON块
        json_match = re.search(r'```json\s*([\s\S]*?)\s*```', response)
        if json_match:
            return json.loads(json_match.group(1))

        # 尝试直接解析
        try:
            return json.loads(response)
        except:
            # 尝试找到数组或对象
            start = response.find('[') if '[' in response else response.find('{')
            end = response.rfind(']') + 1 if ']' in response else response.rfind('}') + 1
            if start >= 0 and end > start:
                return json.loads(response[start:end])

        raise ValueError(f"无法解析LLM响应: {response[:200]}")

    def _extract_title(self, text: str) -> str:
        """提取标题"""
        lines = text.strip().split('\n')
        return lines[0].strip() if lines else "未命名"

    def _load_prompts(self) -> Dict[str, str]:
        """加载提示词模板"""
        return {
            'extract_characters': """分析以下小说片段，提取所有主要角色信息。

- id: 唯一标识 (如 char_001)

            'generate_storyboard': """你是专业的动画分镜师。请将以下小说章节转换为分镜脚本。
